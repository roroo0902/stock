Google Apps Script：Code.gs ========================================================= /** * ===================================================== * 台股代碼查詢 API * Google Apps Script / Code.gs * * 功能： * 1. 接收 ?code=2330 * 2. 同時查詢 TWSE MIS： * tse_2330.tw 上市 * otc_2330.tw 上櫃 * 3. 找到後輸出 JSON * 4. GitHub Pages 前端可呼叫此 Web App * ===================================================== */ function doGet(e) { try { const code = String( (e && e.parameter && e.parameter.code) || '' ) .trim() .toUpperCase(); if (!code) { return jsonOutput_({ ok: false, message: '缺少股票代碼' }); } if (!/^[0-9A-Z]{4,8}$/.test(code)) { return jsonOutput_({ ok: false, message: '股票代碼格式不正確' }); } const stock = lookupTwStock_(code); if (!stock) { return jsonOutput_({ ok: false, message: '找不到此上市／上櫃股票代碼' }); } return jsonOutput_({ ok: true, stock: stock }); } catch (err) { return jsonOutput_({ ok: false, message: String(err && err.message ? err.message : err) }); } } /** * 查詢 TWSE MIS * * 官方行情格式： * tse_2330.tw = 上市 * otc_6488.tw = 上櫃 * * 一次同時送兩個候選， * 回傳後再找真正存在的標的。 */ function lookupTwStock_(code) { const channels = [ 'tse_' + code + '.tw', 'otc_' + code + '.tw' ]; const url = 'https://mis.twse.com.tw/stock/api/getStockInfo.jsp' + '?ex_ch=' + encodeURIComponent(channels.join('|')) + '&json=1' + '&delay=0' + '&_=' + Date.now(); const response = UrlFetchApp.fetch(url, { method: 'get', muteHttpExceptions: true, followRedirects: true, headers: { 'User-Agent': 'Mozilla/5.0 (compatible; GoogleAppsScript)', 'Referer': 'https://mis.twse.com.tw/stock/fibest.jsp' } }); const httpCode = response.getResponseCode(); if (httpCode < 200 || httpCode >= 300) { throw new Error( 'TWSE 查詢失敗，HTTP ' + httpCode ); } const text = response.getContentText('UTF-8'); const data = JSON.parse(text); const list = Array.isArray(data.msgArray) ? data.msgArray : []; /* * API 可能回傳空物件或非該股票資料， * 所以需再確認 c 欄等於使用者輸入代碼。 */ const item = list.find(x => { return ( x && String(x.c || '').toUpperCase() === code && ( String(x.ex || '') === 'tse' || String(x.ex || '') === 'otc' ) ); }); if (!item) { return null; } const exchange = item.ex === 'tse' ? 'TWSE' : 'TPEx'; const marketName = item.ex === 'tse' ? '上市' : '上櫃'; return { code: String(item.c || code), name: String(item.n || '').trim(), fullName: String(item.nf || item.n || '').trim(), exchange: exchange, marketName: marketName, /* * z = 最近成交價 * 若盤中沒有成交，有時會是 "-" */ lastPrice: cleanNumber_(item.z), /* * y = 昨日收盤價 */ previousClose: cleanNumber_(item.y), open: cleanNumber_(item.o), high: cleanNumber_(item.h), low: cleanNumber_(item.l), limitUp: cleanNumber_(item.u), limitDown: cleanNumber_(item.w), volume: cleanNumber_(item.v), quoteDate: String(item.d || ''), quoteTime: String(item.t || ''), rawExchange: String(item.ex || '') }; } /** * 將 "-", 空白等轉成 null。 */ function cleanNumber_(value) { if ( value === null || value === undefined || value === '' || value === '-' ) { return null; } const n = Number(value); return Number.isFinite(n) ? n : null; } /** * JSON Response */ function jsonOutput_(obj) { return ContentService .createTextOutput( JSON.stringify(obj) ) .setMimeType( ContentService.MimeType.JSON ); } /** * 可手動執行測試。 */ function testLookup2330() { const result = lookupTwStock_('2330'); Logger.log( JSON.stringify( result, null, 2 ) ); } /** * 可手動執行測試上櫃股票。 * 若該代碼當下存在於上櫃市場， * 應回傳 marketName = 上櫃。 */ function testLookupOTC() { const result = lookupTwStock_('6488'); Logger.log( JSON.stringify( result, null, 2 ) ); } ========================================================= 部署方式 ========================================================= 【GAS 股票查詢 API】 1. 建立新的 Google Apps Script。 2. 將 B 區 Code.gs 全部貼上。 3. 部署 → 新增部署作業 → 網頁應用程式。 4. 執行身分：我。 5. 誰可以存取：任何人。 6. 複製 Web App 網址。 7. 回到 index.html，找到： const STOCK_LOOKUP_URL = "YOUR_STOCK_LOOKUP_GAS_URL_HERE"; 改成你的 GAS 網址。 【GitHub】 將 A 區內容另存成 index.html，上傳 GitHub Pages。 然後修改： const WEBHOOK_URL = "YOUR_WEBHOOK_URL_HERE"; 改成你的 Make / GAS / n8n Webhook。 ========================================================= Webhook JSON 範例 ========================================================= 輸入： 2330 查詢成功： 台積電 上市 / TWSE 若： 整股 買進 1 張 限價 1000 元 送出內容大致為： { "event": "stock_order", "timestamp": "2026-10-06T13:00:00.000Z", "order": { "stockCode": "2330", "stockName": "台積電", "stockFullName": "台灣積體電路製造股份有限公司", "exchange": "TWSE", "marketName": "上市", "side": "買進", "market": "整股", "session": "整股", "accountType": "現股", "timeInForce": "ROD", "priceType": "限價", "quantity": 1, "quantityUnit": "張", "shares": 1000, "price": 1000, "estimatedAmount": 1000000 }, "marketSnapshot": { "lastPrice": 1000, "previousClose": 995, "quoteDate": "20261006", "quoteTime": "13:30:00" }, "source": { "app": "github-pages-tw-stock-order" } } =========================================================
